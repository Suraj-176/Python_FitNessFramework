  function getRunUrl() {

                         /suite/i.test(document.title) ||
                         document.body.className.indexOf("suite") !== -1;
    if (hasSuiteAction) {
      return p() + "?suite";
    }
    return p() + "?test";
  }

  var NAV = {

  var SESSION_KEY = "ici_session";

  function saveSession(user, role, display) {

  }

  function loadSession() {
    try { return JSON.parse(sessionStorage.getItem(SESSION_KEY)); } catch(e) { return null; }
  }

  }

  /* ── Zero-Dependency Dynamic Table-to-Excel/CSV Exporter! ── */
  window.iciExportTableToCSV = function() {
    var tables = document.querySelectorAll("table.fit_table, table.statistics, .history table, .test-history table, table");
    if (tables.length === 0) {
      alert("No tabular data found on this page to export.");
      return;
    }

    var csvRows = [];
    tables.forEach(function(table, tableIdx) {
      if (tableIdx > 0) csvRows.push(""); // separation spacer
      
      var rows = table.querySelectorAll("tr");
      rows.forEach(function(row) {
        var cols = row.querySelectorAll("th, td");
        var rowData = [];
        
        cols.forEach(function(col) {
          var text = col.textContent.replace(/\s+/g, " ").replace(/"/g, '""').trim();
          rowData.push('"' + text + '"');
        });
        
        csvRows.push(rowData.join(","));
      });
    });

    var csvString = csvRows.join("\n");
    var blob = new Blob([csvString], { type: "text/csv;charset=utf-8;" });
    var link = document.createElement("a");
    
    var pageTitle = (document.title || "Report").replace(/[^a-zA-Z0-9]/g, "_");
    link.download = pageTitle + "_Report.csv";
    link.href = URL.createObjectURL(blob);
    link.style.display = "none";
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

      backBtn.style.color = "#fff";
      backBtn.style.fontWeight = "bold";
      backBtn.textContent = "⬅ Back";
      nav.appendChild(backBtn);

  window.iciLogin = function() {
    var u = (document.getElementById("icici-username").value || "").trim().toLowerCase();
    var p = document.getElementById("icici-password").value || "";
    var errEl = document.getElementById("icici-login-error");

    if (!u || !p) { errEl.textContent = "Please enter username and password."; return; }

    var rec = USERS[u];
    if (!rec || rec.password !== p) {
      errEl.textContent = "Invalid username or password.";
      document.getElementById("icici-password").value = "";
      return;
    }

    errEl.textContent = "";
    saveSession(u, rec.role, rec.display);
    showApp({ display: rec.display, role: rec.role });
  };

  /* Block all FitNesse global keyboard shortcuts & trap focus inside the login box when not logged in */
  (function() {
    var blockHotkeys = function(e) {
      var session = sessionStorage.getItem("ici_session");
      if (!session) {
        var targetId = e.target ? e.target.id : "";
        var isLoginField = (targetId === "icici-username" || targetId === "icici-password" || targetId === "icici-login-btn");
        
        // Let Enter key bubble to trigger login
        if (e.key === "Enter" && isLoginField) {
          return;
        }
        
        // Stop propagation so it never reaches FitNesse's global key event listeners
        e.stopPropagation();
        
        // Prevent default actions (like browser shortcuts, scroll, backspace navigation, etc.) if not in login fields
        if (!isLoginField) {
          e.preventDefault();
        }
      }
    };

    window.addEventListener("keydown", blockHotkeys, true);
    window.addEventListener("keypress", blockHotkeys, true);
    window.addEventListener("keyup", blockHotkeys, true);

    // Focus trap: Redirect focus back to the login box if user tries to focus background elements
    document.addEventListener("focus", function(e) {
      var session = sessionStorage.getItem("ici_session");
      if (!session) {
        var loginBox = document.getElementById("icici-login-box");
        if (loginBox && !loginBox.contains(e.target)) {
          e.stopPropagation();
          var uField = document.getElementById("icici-username");
          if (uField) {
            uField.focus();
          }
        }
      }
    }, true);
  })();

  var STORAGE_KEY = "ici_env_config";
  var DEFAULT_CONFIG = {
    active: "Dev",
    environments: {
      "Dev": "http://192.168.4.187/trubi25v2rel",
      "QA": "https://dummyjson.com",

  var tempEnvironments = {};

  function loadConfig() {
    var stored = localStorage.getItem(STORAGE_KEY);
    if (!stored) {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(DEFAULT_CONFIG));
      return DEFAULT_CONFIG;
    }
    try { return JSON.parse(stored); } catch(e) { return DEFAULT_CONFIG; }
  }

  function saveConfig(config) {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(config));
  }

  function populateEnvUI() {
    var config = loadConfig();
    var select = document.getElementById("ici-nav-env-select");
    if (!select) return;
    
    select.innerHTML = "";
    for (var env in config.environments) {
      var opt = document.createElement("option");
      opt.value = env;
      opt.textContent = env;
      if (env === config.active) opt.selected = true;
      select.appendChild(opt);
    }
    document.getElementById("ici-nav-env-container").style.display = "flex";
  }

  window.iciSwitchEnv = function(selectedEnv) {
    var config = loadConfig();
    if (!config.environments[selectedEnv]) return;
    
    config.active = selectedEnv;
    saveConfig(config);
    
    var activeUrl = config.environments[selectedEnv];
    var formData = new URLSearchParams();
    formData.append("responder", "saveData");
    formData.append("pageContent", "!define BASE_URL {" + activeUrl + "}\n");
    
    fetch("/VariablePage", {
      method: "POST",
      body: formData,
      headers: { "Content-Type": "application/x-www-form-urlencoded" }
    })
    .then(function() {
      location.reload();
    })
    .catch(function(err) {
      console.error("Failed to write to VariablePage:", err);
      location.reload();
    });
  };

  function renderModalTable() {
    var tbody = document.getElementById("ici-modal-table-body");
    if (!tbody) return;
    tbody.innerHTML = "";
    
    for (var env in tempEnvironments) {
      var tr = document.createElement("tr");
      
      var tdName = document.createElement("td");
      tdName.style.fontWeight = "bold";
      tdName.textContent = env;
      tr.appendChild(tdName);
      
      var tdUrl = document.createElement("td");
      var input = document.createElement("input");
      input.type = "text";
      input.className = "ici-env-input";
      input.value = tempEnvironments[env];
      input.setAttribute("data-env", env);
      input.onchange = function(e) {
        var envName = e.target.getAttribute("data-env");
        tempEnvironments[envName] = e.target.value.trim();
      };
      tdUrl.appendChild(input);
      tr.appendChild(tdUrl);
      
      var tdAction = document.createElement("td");
      tdAction.style.textAlign = "center";
      var delBtn = document.createElement("button");
      delBtn.className = "ici-row-btn";
      delBtn.innerHTML = "❌";
      delBtn.setAttribute("data-env", env);
      delBtn.onclick = function(e) {
        var envName = e.target.getAttribute("data-env");
        delete tempEnvironments[envName];
        renderModalTable();
      };
      tdAction.appendChild(delBtn);
      tr.appendChild(tdAction);
      
      tbody.appendChild(tr);
    }
  }

  window.iciSaveEnvironments = function() {
    var addName = (document.getElementById("ici-add-env-name").value || "").trim();
    var addUrl = (document.getElementById("ici-add-env-url").value || "").trim();
    
    if (addName && addUrl) {
      addName = addName.toUpperCase().replace(/\s+/g, "_");
      tempEnvironments[addName] = addUrl;
    }
    
    if (Object.keys(tempEnvironments).length === 0) {
      alert("At least one environment must exist.");
      return;
    }
    
    var config = loadConfig();
    config.environments = tempEnvironments;
    if (!config.environments[config.active]) {
      config.active = Object.keys(tempEnvironments)[0];
    }
    
    saveConfig(config);
    iciCloseManageEnvModal();
    iciSwitchEnv(config.active);
  };

          if (suiteRadio && suiteRadio.checked) {
            textarea.value = "!contents -R2 -g -p -h\n";
          }

              }
            });
          });
